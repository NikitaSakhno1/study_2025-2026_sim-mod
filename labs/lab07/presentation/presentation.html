05 — M/M/c

Реализация обслуживания заявки

@yield request(server)

t_start = now(env)
push!(starts, t_start)

push!(waits, t_start - t_request)

service_time = rand(service_dist)

@yield timeout(env, service_time)

t_finish = now(env)
push!(finishes, t_finish)

@yield unlock(server)
